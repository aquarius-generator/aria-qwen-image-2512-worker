# PINNED WHERE DRIFT WOULD CHANGE THE CONTESTANT, RANGED WHERE A RESOLVER KNOWS BETTER THAN I DO.

# The first build of this worker failed on time. The second built in 7.7 minutes and then every job failed,
# because diffusers was pinned to 0.33.1 while the model's own model_index.json at the pinned revision
# declares `_class_name: QwenImagePipeline` and `_diffusers_version: 0.36.0.dev0`. 0.33.1 contains no
# QwenImagePipeline, no QwenImageTransformer2DModel and no AutoencoderKLQwenImage, so there was nothing for
# the loader to construct.
#
# Reading diffusers 0.42.1's own metadata instead of bumping only the obvious pin found two more that would
# have failed on the next build: it requires huggingface-hub>=1.32 (pinned here at 0.30.2) and
# safetensors>=0.8.0 (pinned here at 0.5.3). Fixing one pin at a time would have cost one 8 minute build
# each to discover.

runpod>=1.7,<2

# torch stays on the line that matches the CUDA 12.4 base image. diffusers 0.42 asks for torch>=2.6, which
# this satisfies, so there is no reason to move it and every reason not to.

# The floor that matters: QwenImagePipeline landed well before this, and 0.42.1 is the current release.
diffusers==0.42.1

# hand-held pin. The floor is where Qwen2_5_VLForConditionalGeneration, the model's text encoder, is
# comfortably available.
transformers>=4.56,<6
accelerate>=0.31

# Required floors, taken from diffusers 0.42.1's metadata rather than guessed.
huggingface-hub>=1.32,<3
safetensors>=0.8.0

sentencepiece
